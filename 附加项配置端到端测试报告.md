   ```javascript
           taskId: 1,

           secretaryParticipation: 1, // 勾选秘书参与
           filledBy: 'admin'
       })
   });
   
   const result = await response.json();
   console.log('保存成功:', result);
   ```

2. **查询附加项**:
   ```javascript
   // 前端调用示例
   const response = await fetch('/api/bonus?taskId=1&institutionId=1');
   const result = await response.json();
   
   // 渲染到界面
   document.getElementById('director').checked = result.data.directorPresentation === 1;
   document.getElementById('secretary').checked = result.data.secretaryParticipation === 1;
   ```

3. **刷新页面**:
